    1991: /**
    1992: * Run a listing query, with its result cached across page loads
    1993: *
    1994: * For queries whose results only change when listings change, such as the values in the
    1995: * search dropdowns. The result is remembered for the rest of the page load and saved as a
    1996: * transient for 12 hours (filter epl_listing_query_cache_expiration).
    1997: *
    1998: * When listings change, visitors keep getting the cached results while a background job
    1999: * (WP-Cron) rebuilds every query visitors have used, then the site switches to the new
    2000: * results, so no visitor waits for the rebuild. If the rebuild hasn't run within 10 minutes
    2001: * (filter epl_listing_query_cache_stale_limit), the next page load rebuilds as it goes.
    2002: *
    2003: * @param string $sql A prepared SQL query.
    2004: * @param string $return_type col for $wpdb->get_col(), results for $wpdb->get_results().
    2005: *
    2006: * @return array
    2007: *
    2008: * @since 3.6
    2009: */
    2010: function epl_listing_query_cache( $sql, $return_type = 'col' ) {
    2011: static $memory = array();
    2012: static $checked = false;
    2013:
    2014: $key = md5( $return_type . '|' . $sql );
    2015:
    2016: if ( isset( $memory[ $key ] ) ) {
    2017: return $memory[ $key ];
    2018: }
    2019:
    2020: /**
    2021: * Filters whether listing query results are cached across page loads.
    2022: *
    2023: * @since 3.6
    2024: *
    2025: * @param bool $enabled Default true.
    2026: */
    2027: $enabled = (bool) apply_filters( 'epl_listing_query_cache_enabled', true );
    2028:
    2029: if ( ! $enabled ) {
    2030: $memory[ $key ] = epl_listing_query_run( $sql, $return_type );
    2031: return $memory[ $key ];
    2032: }
    2033:
    2034: // Listings changed and the background rebuild hasn't run in time: stop serving old results.
    2035: if ( ! $checked ) {
    2036: $checked = true;
    2037: $stale = (int) get_option( 'epl_listing_query_cache_stale', 0 );
    2038:
    2039: /**
    2040: * Filters how long old results may be served after listings change, in seconds,
    2041: * while waiting for the background rebuild.
    2042: *
    2043: * @since 3.6
    2044: *
    2045: * @param int $limit Default 10 minutes.
    2046: */
    2047: if ( $stale && time() - $stale > (int) apply_filters( 'epl_listing_query_cache_stale_limit', 10 * MINUTE_IN_SECONDS ) ) {
    2048: epl_flush_listing_query_cache( true );
    2049: }
    2050: }
    2051:
    2052: $transient = epl_listing_query_cache_transient( $key );
    2053: $result = get_transient( $transient );
    2054:
    2055: if ( ! is_array( $result ) ) {
    2056: $result = epl_listing_query_run( $sql, $return_type );
    2057: set_transient( $transient, $result, epl_listing_query_cache_expiration() );
    2058: epl_listing_query_cache_remember( $key, $sql, $return_type );
    2059: }
    2060:
    2061: $memory[ $key ] = $result;
    2062:
    2063: return $result;
    2064: }
