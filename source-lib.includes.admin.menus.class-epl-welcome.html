    200: public function parse_readme() {
    201: $file = file_exists( EPL_PLUGIN_PATH . 'readme.txt' ) ? EPL_PLUGIN_PATH . 'readme.txt' : null;
    202: if ( ! $file ) {
    203: $readme = '<p>' . esc_html__( 'No valid changelog was found.', 'easy-property-listings' ) . '</p>';
    204: } else {
    205: $readme = file_get_contents( $file ); //phpcs:ignore
    206: $readme = explode( '== Changelog ==', $readme );
    207: $readme = end( $readme );
    208: // Stop at the next section (== Upgrade Notice ==).
    209: $readme = preg_split( '/^== /m', $readme );
    210: $readme = nl2br( esc_html( trim( $readme[0] ) ) );
    211: $readme = preg_replace( '/`(.*?)`/', '<code>\\1</code>', $readme );
    212: $readme = preg_replace( '/[\040]\*\*(.*?)\*\*/', ' <strong>\\1</strong>', $readme );
    213: $readme = preg_replace( '/[\040]\*(.*?)\*/', ' <em>\\1</em>', $readme );
    214: $readme = preg_replace( '/= (.*?) =/', '<h4>\\1</h4>', $readme );
    215: $readme = preg_replace( '/\[(.*?)\]\((.*?)\)/', '<a href="\\2">\\1</a>', $readme );
    216: }
    217: return $readme;
    218: }
